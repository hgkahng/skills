**Plan**
- Load the folder twice with `ImageFolder` (train transforms with augmentation, val transforms without). Split indices once with a seeded permutation, so the split is identical on resume and no augmentation touches val.
- Fine-tune all layers of an ImageNet-pretrained ResNet-18 with its `fc` head replaced. Use SGD with momentum, cosine LR, and fp16 autocast with `GradScaler`.
- Each epoch, write `last.pt` (model, optimizer, scheduler, scaler, epoch, best acc, classes) atomically, and write `best.pt` when val accuracy improves. `--resume` continues from `last.pt`.

```python
"""Fine-tune a pretrained ResNet-18 on an ImageFolder dataset, with AMP and resumable checkpoints.

    python finetune.py /path/to/images
    python finetune.py /path/to/images --resume
"""
import argparse
import random
from pathlib import Path

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader, Subset
from torchvision import datasets, models, transforms as T

MEAN, STD = (0.485, 0.456, 0.406), (0.229, 0.224, 0.225)  # ImageNet stats


def parse_args():
    p = argparse.ArgumentParser()
    p.add_argument("data_dir", type=Path)
    p.add_argument("--out", type=Path, default=Path("runs/resnet18"))
    p.add_argument("--epochs", type=int, default=20)
    p.add_argument("--batch-size", type=int, default=64)
    p.add_argument("--lr", type=float, default=0.01)
    p.add_argument("--weight-decay", type=float, default=1e-4)
    p.add_argument("--val-frac", type=float, default=0.2)
    p.add_argument("--workers", type=int, default=4)
    p.add_argument("--seed", type=int, default=0)
    p.add_argument("--resume", action="store_true", help="continue from <out>/last.pt")
    return p.parse_args()


def save_atomic(path: Path, state: dict) -> None:
    tmp = path.with_suffix(".tmp")
    torch.save(state, tmp)
    tmp.replace(path)  # a crash mid-write never corrupts the previous checkpoint


def run_epoch(model, loader, device, amp, optimizer=None, scaler=None):
    """One pass over `loader`. Trains if `optimizer` is given, otherwise evaluates. Returns (loss, acc)."""
    train = optimizer is not None
    model.train(train)
    loss_sum, correct, total = 0.0, 0, 0
    with (torch.enable_grad() if train else torch.inference_mode()):
        for x, y in loader:
            x, y = x.to(device, non_blocking=True), y.to(device, non_blocking=True)
            with torch.autocast(device.type, dtype=torch.float16, enabled=amp):
                out = model(x)
                loss = F.cross_entropy(out, y)
            if train:
                optimizer.zero_grad(set_to_none=True)
                scaler.scale(loss).backward()
                scaler.step(optimizer)
                scaler.update()
            loss_sum += loss.item() * len(y)
            correct += (out.argmax(1) == y).sum().item()
            total += len(y)
    return loss_sum / total, correct / total


def main():
    args = parse_args()
    random.seed(args.seed)
    np.random.seed(args.seed)
    torch.manual_seed(args.seed)  # also seeds all CUDA devices

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    amp = device.type == "cuda"
    args.out.mkdir(parents=True, exist_ok=True)

    # --- data: two views of the same folder, split by a seeded index permutation ---
    train_tf = T.Compose([T.RandomResizedCrop(224), T.RandomHorizontalFlip(), T.ToTensor(), T.Normalize(MEAN, STD)])
    val_tf = T.Compose([T.Resize(256), T.CenterCrop(224), T.ToTensor(), T.Normalize(MEAN, STD)])
    train_full = datasets.ImageFolder(args.data_dir, train_tf)
    val_full = datasets.ImageFolder(args.data_dir, val_tf)
    perm = torch.randperm(len(train_full), generator=torch.Generator().manual_seed(args.seed)).tolist()
    n_val = int(len(perm) * args.val_frac)
    train_ds, val_ds = Subset(train_full, perm[n_val:]), Subset(val_full, perm[:n_val])

    g = torch.Generator()  # reseeded every epoch so a resumed epoch sees the same shuffle
    loader_kw = dict(batch_size=args.batch_size, num_workers=args.workers, pin_memory=amp,
                     persistent_workers=args.workers > 0)
    train_loader = DataLoader(train_ds, shuffle=True, drop_last=True, generator=g, **loader_kw)
    val_loader = DataLoader(val_ds, **loader_kw)

    # --- model / optimization ---
    num_classes = len(train_full.classes)
    model = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
    model.fc = nn.Linear(model.fc.in_features, num_classes)
    model.to(device)
    optimizer = torch.optim.SGD(model.parameters(), lr=args.lr, momentum=0.9, weight_decay=args.weight_decay)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=args.epochs)
    scaler = torch.amp.GradScaler("cuda", enabled=amp)

    start_epoch, best_acc = 0, 0.0
    if args.resume:
        ckpt = torch.load(args.out / "last.pt", map_location=device)
        model.load_state_dict(ckpt["model"])
        optimizer.load_state_dict(ckpt["optimizer"])
        scheduler.load_state_dict(ckpt["scheduler"])
        scaler.load_state_dict(ckpt["scaler"])
        start_epoch, best_acc = ckpt["epoch"] + 1, ckpt["best_acc"]
        print(f"resumed from epoch {ckpt['epoch']} (best val acc {best_acc:.4f})")

    print(f"{len(train_ds)} train / {len(val_ds)} val images, {num_classes} classes, device={device}, amp={amp}")

    # --- loop ---
    for epoch in range(start_epoch, args.epochs):
        g.manual_seed(args.seed + epoch)
        lr = optimizer.param_groups[0]["lr"]
        tr_loss, tr_acc = run_epoch(model, train_loader, device, amp, optimizer, scaler)
        val_loss, val_acc = run_epoch(model, val_loader, device, amp)
        scheduler.step()

        best_acc = max(best_acc, val_acc)
        state = dict(model=model.state_dict(), optimizer=optimizer.state_dict(), scheduler=scheduler.state_dict(),
                     scaler=scaler.state_dict(), epoch=epoch, best_acc=best_acc, classes=train_full.classes)
        save_atomic(args.out / "last.pt", state)
        if val_acc >= best_acc:
            save_atomic(args.out / "best.pt", state)

        print(f"epoch {epoch:3d} | lr {lr:.5f} | train loss {tr_loss:.4f} acc {tr_acc:.4f} "
              f"| val loss {val_loss:.4f} acc {val_acc:.4f}")


if __name__ == "__main__":
    main()
```

**Notes**
- **Not executed.** No environment was available, so this has not been run or import-checked. Do a one-epoch smoke test (`--epochs 1`) before a long run, and test `--resume` once by interrupting and restarting.
- **Resume.** `--epochs` is the total target, not "additional epochs". The cosine schedule is tied to it, so keep the same value when resuming. Resuming restores the split and per-epoch shuffle order. It does not restore augmentation RNG or cuDNN nondeterminism, so a resumed run is close to, but not bit-identical with, an uninterrupted one.
- **Split.** It is a random split, not stratified. With small or imbalanced classes, per-class val counts can be uneven. The split depends on `--seed` and `--val-frac`, so don't change them when resuming.
- **Class count.** `num_classes` comes from the folder structure, so it will be 10 for your data. If the script reports a different number, you have extra or missing subfolders.
- **AMP.** It uses fp16 autocast with a `GradScaler` on CUDA and runs in fp32 on CPU. On Ampere or newer you could use bf16 and drop the scaler, but the fp16 path matches your request to checkpoint scaler state.
- **Pretrained weights.** `ResNet18_Weights.DEFAULT` downloads from the internet the first time.
- **Hyperparameters.** The defaults (SGD, lr 0.01, 20 epochs) are reasonable starting points for full fine-tuning, not tuned values. If the dataset is small, try a lower lr such as 1e-3 for the backbone.
- **Checkpoints.** `best.pt` and `last.pt` carry the full training state plus `classes`, so either file can be loaded for inference with `model.load_state_dict(ckpt["model"])`.
