Plan:
- One `ImageFolder` is indexed twice (train transforms with augmentation, val transforms without). A seeded random permutation splits the indices, so no val image is augmented and the split is the same on every run.
- Replace the ResNet-18 head with a 10-way layer. Train with `autocast` and `GradScaler` (CUDA only).
- Save model, optimizer, scheduler, scaler and epoch at the end of every epoch. `--resume` picks up from there.

```python
import argparse
import random
from pathlib import Path

import numpy as np
import torch
import torch.nn.functional as F
from torch import nn
from torch.utils.data import DataLoader, Subset
from torchvision import datasets, models, transforms as T


def seed_worker(_):
    seed = torch.initial_seed() % 2**32  # already differs per worker and per epoch
    random.seed(seed)
    np.random.seed(seed)


def main():
    p = argparse.ArgumentParser()
    p.add_argument("data", type=Path, help="ImageFolder root: data/<class>/<image>")
    p.add_argument("--epochs", type=int, default=15)
    p.add_argument("--bs", type=int, default=64)
    p.add_argument("--lr", type=float, default=0.01)
    p.add_argument("--val-frac", type=float, default=0.2)
    p.add_argument("--workers", type=int, default=4)
    p.add_argument("--seed", type=int, default=0)
    p.add_argument("--ckpt", type=Path, default=Path("checkpoint.pt"))
    p.add_argument("--resume", action="store_true")
    a = p.parse_args()

    random.seed(a.seed)
    np.random.seed(a.seed)
    torch.manual_seed(a.seed)  # also seeds every CUDA device

    dev = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    amp = dev.type == "cuda"

    # Data: same files, different transforms. Normalization uses fixed ImageNet stats, not data-fit stats.
    norm = T.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
    train_tf = T.Compose([T.RandomResizedCrop(224), T.RandomHorizontalFlip(), T.ToTensor(), norm])
    val_tf = T.Compose([T.Resize(256), T.CenterCrop(224), T.ToTensor(), norm])
    train_full = datasets.ImageFolder(a.data, train_tf)
    val_full = datasets.ImageFolder(a.data, val_tf)
    perm = torch.randperm(len(train_full), generator=torch.Generator().manual_seed(a.seed)).tolist()
    n_val = int(len(perm) * a.val_frac)
    train_ds, val_ds = Subset(train_full, perm[n_val:]), Subset(val_full, perm[:n_val])

    g = torch.Generator()
    kw = dict(batch_size=a.bs, num_workers=a.workers, pin_memory=amp,
              persistent_workers=a.workers > 0, worker_init_fn=seed_worker, generator=g)
    train_dl = DataLoader(train_ds, shuffle=True, **kw)
    val_dl = DataLoader(val_ds, **kw)

    model = models.resnet18(weights=models.ResNet18_Weights.DEFAULT)
    model.fc = nn.Linear(model.fc.in_features, len(train_full.classes))
    model.to(dev)
    opt = torch.optim.SGD(model.parameters(), lr=a.lr, momentum=0.9, weight_decay=1e-4)
    sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=a.epochs)
    scaler = torch.amp.GradScaler(dev.type, enabled=amp)

    start = 0
    if a.resume and a.ckpt.exists():
        ck = torch.load(a.ckpt, map_location=dev)
        model.load_state_dict(ck["model"])
        opt.load_state_dict(ck["opt"])
        sched.load_state_dict(ck["sched"])
        scaler.load_state_dict(ck["scaler"])
        start = ck["epoch"]
        print(f"resumed from {a.ckpt} at epoch {start}")

    for epoch in range(start, a.epochs):
        g.manual_seed(a.seed + epoch)  # shuffle order and augmentation depend only on (seed, epoch), so resume matches an uninterrupted run
        model.train()
        loss_sum = torch.zeros((), device=dev)
        for x, y in train_dl:
            x, y = x.to(dev, non_blocking=True), y.to(dev, non_blocking=True)
            opt.zero_grad(set_to_none=True)
            with torch.autocast(dev.type, enabled=amp):
                loss = F.cross_entropy(model(x), y)
            scaler.scale(loss).backward()
            scaler.step(opt)
            scaler.update()
            loss_sum += loss.detach() * len(y)
        sched.step()

        model.eval()
        correct = torch.zeros((), device=dev)
        with torch.inference_mode(), torch.autocast(dev.type, enabled=amp):
            for x, y in val_dl:
                x, y = x.to(dev, non_blocking=True), y.to(dev, non_blocking=True)
                correct += (model(x).argmax(1) == y).sum()
        print(f"epoch {epoch + 1}/{a.epochs}  train_loss {loss_sum.item() / len(train_ds):.4f}"
              f"  val_acc {correct.item() / len(val_ds):.4f}")

        tmp = a.ckpt.with_suffix(".tmp")  # write then rename, so a crash mid-save can't corrupt the checkpoint
        torch.save({"model": model.state_dict(), "opt": opt.state_dict(), "sched": sched.state_dict(),
                    "scaler": scaler.state_dict(), "epoch": epoch + 1}, tmp)
        tmp.replace(a.ckpt)


if __name__ == "__main__":
    main()
```

Usage: `python train.py path/to/data`, then `python train.py path/to/data --resume` after an interruption.

Things to know:
- **Not run.** No environment was available, so I have not run or imported this. Do a one-epoch dry run on a small folder first.
- **Same arguments on resume.** The split depends on `--seed` and `--val-frac`, and the cosine schedule depends on `--epochs`. Resume with the same values, or the val set will change and leak into training, and the schedule will be wrong.
- **Reproducible, not deterministic.** All RNGs are seeded, but cuDNN autotuning and some CUDA kernels can still differ between runs. For bit-exact runs, add `torch.backends.cudnn.deterministic = True` and `torch.backends.cudnn.benchmark = False`, at some cost in speed.
- **Mixed precision runs only on CUDA.** On CPU the autocast and scaler are disabled and the script runs in fp32. `torch.amp.GradScaler("cuda", ...)` needs torch 2.3 or newer.
- **Val is your only held-out set.** If you tune hyperparameters on val accuracy, that number is optimistic. Keep a separate test folder for the final number.
- **Hyperparameters are starting points.** `lr=0.01`, SGD with cosine decay and 15 epochs are common fine-tuning defaults. I did not tune them for your data.
- **Checkpoint scope.** It stores the last epoch only, not the best epoch. Say if you want a best-by-val copy as well.
